# One argv per line, split on whitespace, so every token must be free of
# spaces. Chosen to cross every verdict shape the engine has — allow, allow with
# confirmation, and each denial class — on the rules that actually ship. Every
# row of the matrix must produce byte-identical answers for all of them; the
# backend underneath is not a policy input.

# svc-control: no shipped rule restarts a service. The first row used to match
# the lifecycle rule; it must now be refused with the other near misses.

/usr/bin/systemctl restart nginx.service
/usr/bin/systemctl --no-pager --no-ask-password restart nginx
/usr/bin/systemctl --no-pager --no-ask-password restart nginx.service --now
/usr/bin/systemctl --no-pager --no-ask-password enable nginx.service
